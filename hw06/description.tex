\subsubsection*{(i) Exact result}
The exact value is
\[
\frac{4}{5}+\frac{1}{3}=\frac{17}{15}=1.1\overline{3}.
\]

\subsubsection*{(ii) Three-digit chopping}
After chopping each summand to three significant digits,
\[
\frac{4}{5}+\frac{1}{3}=0.800+0.333=1.13.
\]
Thus, the relative error is
\[
e=\left|\frac{1.13-\frac{17}{15}}{\frac{17}{15}}\right|
  \approx 0.002941.
\]

\subsubsection*{(iii) Three-digit rounding}
The rounded result is the same in this case:
\[
\frac{4}{5}+\frac{1}{3}=0.800+0.333=1.13.
\]
Hence,
\[
e=\left|\frac{1.13-\frac{17}{15}}{\frac{17}{15}}\right|
  \approx 0.002941.
\]

\subsection*{b}
\subsubsection*{(i) Exact result}
The exact value is
\[
\left(\frac{1}{3}+\frac{3}{11}\right)-\frac{3}{20}=\frac{301}{660}.
\]

\subsubsection*{(ii) Three-digit chopping}
Chopping each fraction to three significant digits gives
\[
\left(\frac{1}{3}+\frac{3}{11}\right)-\frac{3}{20}
=(0.333+0.272)-0.150=0.455.
\]
Therefore,
\[
e=\left|\frac{\frac{301}{660}-0.455}{\frac{301}{660}}\right|
  \approx 0.002326.
\]

\subsubsection*{(iii) Three-digit rounding}
Rounding each intermediate value yields
\[
\left(\frac{1}{3}+\frac{3}{11}\right)-\frac{3}{20}
=(0.333+0.273)-0.150=0.456.
\]
The corresponding relative error is
\[
e=\left|\frac{\frac{301}{660}-0.456}{\frac{301}{660}}\right|
  \approx 1.3289\times 10^{-4}.
\]
